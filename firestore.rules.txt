
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // A user can read their own user document.
    // They can create their own user document.
    // They can only update or delete their own user document.
    match /users/{userId} {
      allow read, create: if request.auth != null;
      allow update, delete: if request.auth != null && request.auth.uid == userId;
    }

    // Users can only manage their own skill profiles.
    match /users/{userId}/skillProfiles/{profileId} {
        allow read, write, delete: if request.auth != null && request.auth.uid == userId;
    }

    // Users can only manage their own saved proposals.
    match /users/{userId}/savedProposals/{proposalId} {
        allow read, write, delete: if request.auth != null && request.auth.uid == userId;
    }
  }
}
