rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // This rule allows an authenticated user to read, update, or delete their own user document,
    // and allows any authenticated user to create a user document (necessary for sign-up).
    match /users/{userId} {
      allow read, update, delete: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null;

      // This rule allows an authenticated user to perform all actions
      // (create, read, update, delete, list) on their own portfolioCollections.
      match /portfolioCollections/{collectionId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      // This rule allows an authenticated user to perform all actions
      // (create, read, update, delete, list) on their own savedProposals.
      match /savedProposals/{proposalId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }
  }
}
